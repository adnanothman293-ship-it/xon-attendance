  } catch(e){ bigDone(false, 'لم يتم', errText(e)); pb.disabled = false; }
  finally { punching = false; }
}

/* ================== الجدول الشهري ================== */
async function monthScreen(workerId, y, mo, selfView){
  stopClock();
  const now = bgParts(new Date());
  y = y || now.y; mo = mo || now.m;
  if (current){ current.args = [workerId, y, mo, selfView]; }
  app.innerHTML = topbar(selfView ? 'سجلي الشهري' : 'بيانات العامل', true) + '<div class="wrap"><div class="spin"></div></div>';
  const from = `${y}-${pad(mo)}-01`, to = `${y}-${pad(mo)}-${pad(daysInMonth(y, mo))}`;
  let w, logs, settings, email = '';
  try {
    const [wr, lr] = await Promise.all([
      sb.from('att_workers').select('*').eq('id', workerId).single(),
      sb.from('att_logs').select('*').eq('worker_id', workerId).gte('work_date', from).lte('work_date', to).order('check_in')
    ]);
    if (wr.error) throw wr.error; if (lr.error) throw lr.error;
    w = wr.data; logs = lr.data || [];
    settings = (ME && ME.settings) || { daily_hours: 9, weekend_dow: [5] };
    if (!selfView){ const c = workersCache.find(x => x.id === workerId); email = c ? c.email : ''; }
  } catch(e){ app.querySelector('.wrap').innerHTML = `<div class="card msg err show">${esc(errText(e))}</div>`; return; }

  const daily = Number(settings.daily_hours) * 60;
  const weekend = settings.weekend_dow || [5];
  const tStr = todayStr();
  const createdStr = (() => { const p = bgParts(new Date(w.created_at)); return `${p.y}-${pad(p.m)}-${pad(p.d)}`; })();
  const endH = settings.work_end ? +String(settings.work_end).slice(0,2) : 18, endM = settings.work_end ? +String(settings.work_end).slice(3,5) : 0;
  const pastEndToday = (now.H * 60 + now.M) >= (endH * 60 + endM);

  let expected = 0, actual = 0, short = 0, over = 0, workdays = 0, presentDays = 0;
  const rows = [];
  for (let d = 1; d <= daysInMonth(y, mo); d++){
    const ds = `${y}-${pad(mo)}-${pad(d)}`, dow = dowOf(ds), isOff = weekend.includes(dow);
    const dl = logs.filter(l => l.work_date === ds);
    let mins = 0, openFlag = false;
    dl.forEach(l => {
      if (l.check_out) mins += (new Date(l.check_out) - new Date(l.check_in)) / 60000;
      else if (ds === tStr) { mins += (Date.now() - new Date(l.check_in)) / 60000; openFlag = true; }
      else openFlag = true;
    });
    mins = Math.max(0, mins);
    const beforeStart = ds < createdStr;
    let dShort = 0, dOver = 0;
    if (!isOff){
      if (!beforeStart){ expected += daily; workdays++; }
      if (mins > daily) dOver = mins - daily;
      const dayDone = ds < tStr || (ds === tStr && pastEndToday && !dl.some(l => !l.check_out));
      if (!beforeStart && dayDone && mins < daily) dShort = daily - mins;
    } else dOver = mins;
    if (dl.length) presentDays++;
    actual += mins; short += dShort; over += dOver;
    rows.push({ ds, d, dow, isOff, dl, mins, openFlag, dShort, dOver, beforeStart });
  }

  EDIT_W = { id: w.id, email, full_name: w.full_name, department: w.department, is_admin: w.is_admin, active: w.active };
  const html = `
  <div class="print-only" style="text-align:center;margin-bottom:8px"><h2>XON — تقرير الحضور والانصراف</h2></div>
  <div class="card">
    <div class="row" style="align-items:flex-start">
      <div class="av" style="flex:0 0 54px;width:54px;height:54px;font-size:18px">${esc(initials(w.full_name))}</div>
      <div style="flex:1">
        <h2 style="margin:0">${esc(w.full_name)}</h2>
        <div class="mute">القسم: <b style="color:var(--ink)">${esc(w.department || '—')}</b></div>
        ${email ? `<div class="mute num" style="text-align:right">${esc(email)}</div>` : ''}
        ${!w.active ? '<span class="badge b-mute">موقوف</span>' : ''}
      </div>
    </div>
    ${!selfView ? `<div class="row noprint" style="margin-top:6px">
      <button class="btn light" style="margin-top:6px" onclick="workerForm(EDIT_W)">✏️ تعديل البيانات</button>
      <button class="btn light" style="margin-top:6px" onclick="window.print()">🖨️ طباعة</button></div>` : `<button class="btn light noprint" onclick="window.print()">🖨️ طباعة</button>`}
  </div>
  <div class="card">
    <div class="monthbar">
      <button onclick="navMonth('${workerId}',${y},${mo},1,${!!selfView})">›</button>
      <div class="m">${AR_MONTHS[mo-1]} ${y}</div>
      <button onclick="navMonth('${workerId}',${y},${mo},-1,${!!selfView})">‹</button>
    </div>
    ${!selfView ? '<div class="mute noprint" style="margin-bottom:8px">اضغط على أي يوم لتعديل أو إضافة دخول/خروج يدوياً</div>' : ''}
    <div class="tblwrap"><table>
      <thead><tr><th>اليوم</th><th>دخول</th><th>خروج</th><th>ساعات العمل</th></tr></thead>
      <tbody>${rows.map(r => `
        <tr class="${r.isOff ? 'fri' : ''} ${r.ds === tStr ? 'today' : ''} ${!selfView ? 'click' : ''}" ${!selfView ? `onclick="dayEditor('${workerId}','${r.ds}')"` : ''}>
          <td>${AR_DAYS[r.dow]}<span class="sub num" style="font-size:11px;color:var(--mute)">${r.d}/${mo}</span></td>
          <td>${r.dl.length ? r.dl.map(l => `<span class="sub num">${hm(l.check_in)}</span>`).join('') : (r.isOff ? '<span class="badge b-warn">عطلة</span>' : (r.ds > tStr || r.beforeStart || (r.ds === tStr && !pastEndToday)) ? '—' : '<span class="badge b-bad">غياب</span>')}</td>
          <td>${r.dl.map(l => `<span class="sub num">${l.check_out ? hm(l.check_out) : (r.ds === tStr ? '<span class="badge b-ok">داخل</span>' : '<span class="badge b-warn">بدون خروج</span>')}</span>`).join('') || '—'}</td>
          <td class="num"><b>${r.dl.length ? fmtH(r.mins) : '—'}</b>
            ${r.dOver > 0.5 ? `<span class="sub" style="color:var(--ok);font-size:12px">+${fmtH(r.dOver)}</span>` : ''}
            ${r.dShort > 0.5 && r.dl.length ? `<span class="sub" style="color:var(--bad);font-size:12px">-${fmtH(r.dShort)}</span>` : ''}</td>
        </tr>`).join('')}
      </tbody>
      <tfoot>
        <tr class="sumrow"><td colspan="3">ساعات العمل المتوقعة (${workdays} يوم × ${settings.daily_hours} ساعة)</td><td class="num">${fmtH(expected)}</td></tr>
        <tr class="sumrow"><td colspan="3">ساعات العمل الفعلية (${presentDays} يوم حضور)</td><td class="num">${fmtH(actual)}</td></tr>
        <tr class="sumrow"><td colspan="3" style="color:var(--bad)">النقص</td><td class="num" style="color:var(--bad)">${fmtH(short)}</td></tr>
        <tr class="sumrow"><td colspan="3" style="color:var(--ok)">الإضافي</td><td class="num" style="color:var(--ok)">${fmtH(over)}</td></tr>
      </tfoot>
    </table></div>
    <div class="sumgrid">
      <div class="stat s1"><div class="v">${fmtH(expected)}</div><div class="k">المتوقعة</div></div>
      <div class="stat s2"><div class="v">${fmtH(actual)}</div><div class="k">الفعلية</div></div>
      <div class="stat s3"><div class="v">${fmtH(short)}</div><div class="k">النقص</div></div>
      <div class="stat s4"><div class="v">${fmtH(over)}</div><div class="k">الإضافي</div></div>
    </div>
    <div class="mute" style="margin-top:10px">النقص يُحسب لكل يوم دوام انتهى ولم يكمل فيه العامل ${settings.daily_hours} ساعات (والغياب يُحسب نقصاً كاملاً). الإضافي = ما زاد عن ${settings.daily_hours} ساعات في يوم الدوام + كل ساعات أيام العطلة.</div>
  </div>`;
  app.querySelector('.wrap').innerHTML = html;
}
function navMonth(id, y, m, delta, selfView){ m += delta; if (m < 1){ m = 12; y--; } if (m > 12){ m = 1; y++; } current.args = [id, y, m, selfView]; monthScreen(id, y, m, selfView); }

/* ---------- تعديل يوم (للمدير) ---------- */

  const { data: logs, error } = await sb.from('att_logs').select('*').eq('worker_id', workerId).eq('work_date', ds).order('check_in');
  if (error) return toast(errText(error));
  const m = document.createElement('div'); m.className = 'modal';
  const blank = `${ds}T`;
  const block = (l, i) => `<div class="card" style="box-shadow:none;border:1px solid var(--line);padding:12px" data-id="${l ? l.id : ''}">
      <div class="row"><b>${l ? 'فترة ' + (i + 1) : 'إضافة فترة جديدة'}</b>${l && l.in_ip === 'manual' ? '<span class="badge b-warn" style="flex:0">يدوي</span>' : ''}</div>
      <div class="row"><div><label>الدخول</label><input type="datetime-local" class="e_in" value="${l ? localInputVal(l.check_in) : blank + '09:00'}"></div>
      <div><label>الخروج</label><input type="datetime-local" class="e_out" value="${l && l.check_out ? localInputVal(l.check_out) : (l ? '' : blank + '18:00')}"></div></div>
      <label>ملاحظة</label><input class="e_note" value="${esc(l?.note || '')}" placeholder="اختياري">
      <div class="row"><button class="btn e_save" style="margin-top:8px">${l ? 'حفظ' : 'إضافة'}</button>${l ? '<button class="btn red e_del" style="margin-top:8px">حذف</button>' : ''}</div>
    </div>`;
  const [yy, mm, dd] = ds.split('-');
  m.innerHTML = `<div class="sheet"><h2>${AR_DAYS[dowOf(ds)]} ${+dd}/${+mm}/${yy}</h2>
    ${(logs || []).map(block).join('')}${block(null)}
    <div id="e_m" class="msg err"></div>
    <button class="btn light" onclick="this.closest('.modal').remove()">إغلاق</button></div>`;
  document.body.appendChild(m);
  m.addEventListener('click', e => { if (e.target === m) m.remove(); });
  const err = t => { const x = m.querySelector('#e_m'); x.textContent = t; x.className = 'msg err show'; };
  m.querySelectorAll('[data-id]').forEach(box => {
    const id = box.dataset.id ? +box.dataset.id : null;
    box.querySelector('.e_save').onclick = async () => {
      const i = box.querySelector('.e_in').value, o = box.querySelector('.e_out').value;
      if (!i) return err('حدد وقت الدخول');
      try { await rpc('att_admin_save_log', { p_id: id, p_worker: workerId, p_in: inputToIso(i), p_out: inputToIso(o), p_note: box.querySelector('.e_note').value });
        m.remove(); toast('✅ تم الحفظ'); refreshScreen(); }
      catch(e){ err(errText(e)); }
    };
    const del = box.querySelector('.e_del');
    if (del) del.onclick = async () => {
      if (!confirm('حذف هذه الفترة؟')) return;
      const { error } = await sb.from('att_logs').delete().eq('id', id);
      if (error) return err(errText(error));
      m.remove(); toast('تم الحذف'); refreshScreen();
    };
  });
}
